##############################################################################
# employee_capacity.py
#
#  • Loads three workbooks (STR, CoreAlm, Milestone) that live in separate
#    .xlsx files.
#  • Aggregates STR to one row per employee × Monday-week.
#  • Uses dual weekly-capacity rules:
#        < 2025-01-01   → 43 h nominal week
#        ≥ 2025-01-01   → 40 h nominal week
#  • Flags employees who will be freed within the 30-day horizon because a
#    project they last touched has a *_Completed milestone inside that window.
#  • Trains an XGBoost model (falls back to Random-Forest if xgboost missing).
#  • Forecasts **four successive weeks** (Mon–Fri blocks) from RUN_DATE.
#  • Computes capacity gaps each week vs. 94 % utilisation target.
#  • Divides the gap by the **median** historical project weekly load
#    (derived from STR) → new projects needed per week and in total.
#  • Prints a compact summary table.
#
# NB:  edit STR_PATH / CORE_PATH / MILE_PATH below and run:  python employee_capacity.py
##############################################################################

from __future__ import annotations
import logging, re, sys
from pathlib import Path
from datetime import datetime, timedelta

import numpy as np
import pandas as pd
from sklearn.model_selection import TimeSeriesSplit
from sklearn.metrics import mean_absolute_error

try:
    from xgboost import XGBRegressor
    MODEL_ALGO = "xgb"
except ImportError:
    from sklearn.ensemble import RandomForestRegressor
    MODEL_ALGO = "rf"

# ──────────────────────────── HARD-CODED PATHS ─────────────────────────———
STR_PATH  = Path(r"C:\data\STR_cleaned.xlsx")          # ← change me
CORE_PATH = Path(r"C:\data\CoreAlm_master.xlsx")       # ← change me
MILE_PATH = Path(r"C:\data\Milestone_tracker.xlsx")    # ← change me

RUN_DATE      = pd.Timestamp("2025-06-03")             # horizon anchor
HORIZON_DAYS  = 30                                     # 1 month
UTIL_TARGET   = 0.94
HOURS_PRE2025 = 43
HOURS_2025ON  = 40
LAG_W         = (1, 2, 3, 4)
ROLL_W        = (4, 8, 12)

# column aliases (rename here if your sheets differ)
COL_EMAIL   = "Resource_Email"
COL_HOURS   = "Total_Hours_By_week"
COL_WEEK    = "Week"
COL_PROJECT = "ProjectCode"

# ────────────────────────── SET UP LOGGING ──────────────────────────────────
logging.basicConfig(level=logging.INFO, format="%(levelname)s: %(message)s")

# ────────────────────────── SMALL HELPERS ───────────────────────────────────
ISO_PATTERN = re.compile(r"^(\d{4})[-_]?W?(\d{2})$", re.I)

def parse_week(val):
    """Return a Monday Timestamp or NaT for bad strings."""
    if pd.isna(val):
        return pd.NaT
    d = pd.to_datetime(val, errors="coerce")
    if not pd.isna(d):
        return d.normalize()
    m = ISO_PATTERN.match(str(val).strip())
    if m:
        y, w = m.groups()
        return pd.to_datetime(f"{y}-W{w}-1", format="%G-W%V-%u", errors="coerce")
    return pd.NaT

def iso_cols(df, col="Week"):
    iso = df[col].dt.isocalendar()
    if isinstance(iso, pd.DataFrame):          # pandas ≥1.1
        df["ISO_Year"]   = iso["year"].astype("float32")
        df["WeekOfYear"] = iso["week"].astype("float32")
    else:                                      # very old pandas
        df["ISO_Year"]   = [t[0] for t in iso]
        df["WeekOfYear"] = [t[1] for t in iso]
    return df

def add_lags_rolls(f, lags=LAG_W, rolls=ROLL_W):
    for l in lags:
        f[f"lag{l}"] = f.groupby("Employee")["TotalHours"].shift(l)
    for r in rolls:
        f[f"roll{r}"] = (f.groupby("Employee")["TotalHours"]
                           .shift(1).rolling(r, min_periods=1).mean())
    return f

def load_xlsx(path: Path) -> pd.DataFrame:
    if not path.exists():
        logging.error("File not found: %s", path)
        sys.exit(1)
    return pd.read_excel(path)

# ────────────────────────── 1. LOAD THREE SHEETS ────────────────────────────
df_str  = load_xlsx(STR_PATH)
df_core = load_xlsx(CORE_PATH)
df_mile = load_xlsx(MILE_PATH)

# ────────────────────────── 2. PRE-CLEAN STR ───────────────────────────────
df_str[COL_WEEK] = df_str[COL_WEEK].apply(parse_week)
df_str = df_str[df_str[COL_WEEK].notna()].copy()

# median project weekly load before dropping ProjectCode
proj_week = (df_str.groupby([COL_PROJECT, COL_WEEK], as_index=False)
                     [COL_HOURS].sum()
                     .rename(columns={COL_HOURS:"Hours"}))
MED_PROJ_WK_H = proj_week.groupby(COL_PROJECT)["Hours"].mean().median()
logging.info("Median project weekly hours = %.1f", MED_PROJ_WK_H)

# aggregate to Employee × Monday-week
df_str = (df_str.groupby([COL_EMAIL, COL_WEEK], as_index=False)
                 .agg(TotalHours=(COL_HOURS,"sum"))
                 .rename(columns={COL_EMAIL:"Employee", COL_WEEK:"Week"}))

# add nominal capacity
df_str["NominalCap"] = np.where(df_str["Week"] < pd.Timestamp("2025-01-01"),
                                HOURS_PRE2025, HOURS_2025ON)

# ────────────────────────── 3. FLAG EMPLOYEES FREED BY MILESTONE ───────────
close_cols = [c for c in df_mile.columns if c.lower().endswith("_completed")]
cutoff_dt  = RUN_DATE + timedelta(days=HORIZON_DAYS)
closing_proj = (df_mile.loc[df_mile[close_cols].lt(cutoff_dt).any(axis=1),
                            "ProjectCode"].unique())
closing_emp  = (df_core[df_core["ProjectSTRCode"].isin(closing_proj)]
                        ["AssignedTo"].unique())
df_str["WillFree"] = df_str["Employee"].isin(closing_emp).astype("float32")

# ────────────────────────── 4. FEATURE TABLE & MODEL —──────────────────────
df_feat = add_lags_rolls(df_str.copy())
df_feat = iso_cols(df_feat)
emp2id  = {e:i for i,e in enumerate(df_feat["Employee"].unique())}
df_feat["EmpID"] = df_feat["Employee"].map(emp2id).astype("float32")

feat_cols = ["ISO_Year","WeekOfYear","EmpID","NominalCap","WillFree"] + \
            [c for c in df_feat.columns if c.startswith(("lag","roll"))]

df_feat = df_feat.dropna(subset=feat_cols+["TotalHours"])
df_feat[feat_cols] = df_feat[feat_cols].astype("float32")

X, y = df_feat[feat_cols], df_feat["TotalHours"].astype("float32")

if MODEL_ALGO == "xgb":
    from xgboost import XGBRegressor
    model = XGBRegressor(n_estimators=400, learning_rate=0.05,
                         max_depth=6, subsample=0.8, colsample_bytree=0.8,
                         objective="reg:squarederror", n_jobs=-1, random_state=42)
else:
    from sklearn.ensemble import RandomForestRegressor
    model = RandomForestRegressor(n_estimators=400, n_jobs=-1, random_state=42)

mae = [mean_absolute_error(y.iloc[val],
                           model.fit(X.iloc[tr], y.iloc[tr]).predict(X.iloc[val]))
       for tr,val in zip(*list(zip(*list(TimeSeriesSplit(5).split(X)))))]
logging.info("CV MAE ≈ %.2f h", np.mean(mae))
model.fit(X, y)

# ────────────────────────── 5. BUILD 4-WEEK FORECAST ───────────────────────
blocks = []
latest = df_feat.sort_values(["Employee","Week"]).groupby("Employee").tail(1)

for idx in range(4):    # 4 successive Monday blocks
    base = latest.copy() if idx == 0 else blocks[-1].copy()
    base["Week"] = (base["Week"] + pd.Timedelta(days=7)).normalize()
    for l in LAG_W:
        base[f"lag{l}"] = base["TotalHours"] if l == 1 else base[f"lag{l-1}"]
    base = iso_cols(base)
    base["EmpID"] = base["Employee"].map(emp2id).fillna(-1).astype("float32")
    base[feat_cols] = base[feat_cols].astype("float32")
    base["TotalHours"] = model.predict(base[feat_cols]).astype("float32")
    base["WeekIdx"] = idx + 1
    blocks.append(base)

fc = pd.concat(blocks, ignore_index=True)

# ────────────────────────── 6. GAP & PROJECTS NEEDED ───────────────────────
summary = []
for wk in range(1, 5):
    wk_df = fc[fc["WeekIdx"] == wk]
    cap   = wk_df["NominalCap"].sum()
    demand= wk_df["TotalHours"].sum()
    gap   = max(0, UTIL_TARGET * cap - demand)
    need  = int(np.ceil(gap / MED_PROJ_WK_H))
    summary.append((wk, cap, demand, gap, need))

# ────────────────────────── 7. REPORT ───────────────────────────────────────
print(f"\nHORIZON : {RUN_DATE.date()} → {(RUN_DATE + timedelta(days=HORIZON_DAYS)).date()}")
print("Wk |   Capacity |   Demand |     Gap | NewProj")
print("-----------------------------------------------")
for wk, cap, dem, gap, need in summary:
    print(f"{wk:>2d} | {cap:10,.0f} | {dem:8,.0f} | {gap:8,.0f} | {need:7d}")
print(f"\nTOTAL projects needed next 4 weeks : {sum(s[-1] for s in summary)}")
