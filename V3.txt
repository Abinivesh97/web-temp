# capacity_forecast_realdata.py  ──────────────────────────────────────────
import pandas as pd
import numpy as np
from pathlib import Path
from datetime import timedelta
from pandas.tseries.offsets import MonthBegin
from sklearn.model_selection import TimeSeriesSplit
from sklearn.ensemble import RandomForestRegressor
from sklearn.metrics import mean_absolute_error
import warnings, re

warnings.filterwarnings("ignore")

# ────────────────────────── 1. CONFIG  ───────────────────────────────────
STR_FILE   = Path(r"C:\Users\aseneiva\Documents\python scripts\pmo\str_cleaned.xlsx")
CORE_FILE  = Path(r"C:\Users\aseneiva\Documents\python scripts\pmo\datasheets\corealm_export\Master_sheet.xlsx")

STR_SHEET  = "Sheet1"   # tab name with weekly hours
CORE_SHEET = 0          # first sheet

COL_EMAIL  = "Resource_Email"
COL_HOURS  = "Total_Hours_By_week"
COL_WEEK   = "Week"

UTILISATION_TARGET = 0.94          # 94 %
HOURS_PER_WEEK     = 40            # 8 h × Mon-Fri
N_WEEKS_IN_MONTH   = 4             # still used for capacity, not for forecast

# ────────────────────────── 2. LOAD & CLEAN  ────────────────────────────
def parse_week(val):
    """Return Timestamp for Monday of that week or given date; else NaT."""
    if pd.isna(val):                          # missing
        return pd.NaT
    txt = str(val).strip()
    # 1) try direct date
    d = pd.to_datetime(txt, errors="coerce")
    if not pd.isna(d):
        return d.normalize()
    # 2) ISO week like 2023_W08 or 2023-W08
    m = re.match(r"^(\d{4})[-_]?W?(\d{2})$", txt, flags=re.I)
    if m:
        y, w = m.groups()
        return pd.to_datetime(f"{y}-W{w}-1", format="%G-W%V-%u", errors="coerce")
    return pd.NaT

df_str = pd.read_excel(STR_FILE, sheet_name=STR_SHEET)
df_core = pd.read_excel(CORE_FILE, sheet_name=CORE_SHEET)

df_str[COL_WEEK] = df_str[COL_WEEK].apply(parse_week)
df_str = df_str[df_str[COL_WEEK].notna()].copy()   # drop bad rows
df_str = (
    df_str[[COL_EMAIL, COL_HOURS, COL_WEEK]]
    .rename(columns={COL_EMAIL: "ResourceEmail",
                     COL_HOURS: "TotalHours",
                     COL_WEEK:  "Week"})
    .sort_values(["ResourceEmail", "Week"])
    .reset_index(drop=True)
)

# ─────────── 3. FEATURE HELPERS ─────────────────────────────────────────
def add_lags_rolls(frame, lags=(1,2,3,4), rolls=(4,8,12)):
    for l in lags:
        frame[f"lag{l}"] = frame.groupby("ResourceEmail")["TotalHours"].shift(l)
    for r in rolls:
        frame[f"roll{r}"] = (
            frame.groupby("ResourceEmail")["TotalHours"]
            .shift(1).rolling(r, min_periods=1).mean()
        )
    return frame

def add_iso_cols(frame, date_col="Week"):
    iso = frame[date_col].dt.isocalendar()
    if isinstance(iso, pd.DataFrame):         # pandas ≥1.1
        frame["ISO_Year"]   = iso["year"].astype("float32")
        frame["WeekOfYear"] = iso["week"].astype("float32")
    else:                                     # very old pandas
        frame["ISO_Year"]   = [t[0] for t in iso]
        frame["WeekOfYear"] = [t[1] for t in iso]
        frame[["ISO_Year","WeekOfYear"]] = frame[["ISO_Year","WeekOfYear"]].astype("float32")
    return frame

# ─────────── 4. BUILD FEATURES & TRAIN MODEL ───────────────────────────
df_feat = add_lags_rolls(df_str.copy())
df_feat = add_iso_cols(df_feat, "Week")

emp2id = {e:i for i, e in enumerate(df_feat["ResourceEmail"].unique())}
df_feat["EmpID"] = df_feat["ResourceEmail"].map(emp2id).astype("float32")

X_cols = (
    ["WeekOfYear", "ISO_Year", "EmpID"] +
    [c for c in df_feat.columns if c.startswith(("lag","roll"))]
)

df_feat = df_feat.dropna(subset=X_cols + ["TotalHours"]).copy()
df_feat[X_cols] = df_feat[X_cols].astype("float32")
y = df_feat["TotalHours"].astype("float32")
X = df_feat[X_cols]

tscv  = TimeSeriesSplit(n_splits=5)
model = RandomForestRegressor(n_estimators=400, n_jobs=-1, random_state=42)

mae_scores = []
for tr_idx, val_idx in tscv.split(X):
    model.fit(X.iloc[tr_idx], y.iloc[tr_idx])
    pred_val = model.predict(X.iloc[val_idx])
    mae_scores.append(mean_absolute_error(y.iloc[val_idx], pred_val))
print(f"Average CV MAE: {np.mean(mae_scores):.2f} h")

model.fit(X, y)      # full-fit

# ─────────── 5. NEXT-WEEK / NEXT-MONTH FORECAST ────────────────────────
latest = (
    df_feat.sort_values(["ResourceEmail","Week"])
           .groupby("ResourceEmail").tail(1).copy()
)

def next_week_row(r):
    new = r.copy()
    new["Week"] = (r["Week"] + timedelta(days=7)).normalize()   # next Monday
    for l in (1,2,3,4):
        new[f"lag{l}"] = r["TotalHours"] if l==1 else r[f"lag{l-1}"]
    return new

future_rows = latest.apply(next_week_row, axis=1)
future_rows = add_iso_cols(future_rows, "Week")
future_rows["EmpID"] = future_rows["ResourceEmail"].map(emp2id).fillna(-1).astype("float32")
future_rows[X_cols] = future_rows[X_cols].astype("float32")

future_rows["PredHoursNextWeek"] = model.predict(future_rows[X_cols]).astype("float32")

# ---- scale to calendar-exact Mon–Fri work-days in coming month ----------
start_dates = future_rows["Week"]                    # next Monday
end_dates   = start_dates + MonthBegin(1)            # first work-day of next month
workdays = np.busday_count(
    start_dates.dt.date.values.astype("datetime64[D]"),
    end_dates.dt.date.values.astype("datetime64[D]")
)
future_rows["WorkDaysNextMonth"] = workdays
future_rows["PredHoursNextMonth"] = future_rows["PredHoursNextWeek"] * (workdays / 5)

# ─────────── 6. CAPACITY & GAP ─────────────────────────────────────────
n_emp          = future_rows.shape[0]
capacity_week  = n_emp * HOURS_PER_WEEK
capacity_month = capacity_week * N_WEEKS_IN_MONTH
target_week    = UTILISATION_TARGET * capacity_week
target_month   = UTILISATION_TARGET * capacity_month

forecast_week_hours  = future_rows["PredHoursNextWeek"].sum()
forecast_month_hours = future_rows["PredHoursNextMonth"].sum()

gap_week  = max(0, target_week  - forecast_week_hours)
gap_month = max(0, target_month - forecast_month_hours)

print("\n—— Utilisation forecast ——")
print(f"Predicted total next Mon–Fri : {forecast_week_hours:,.0f} h")
print(f"94 % capacity target week   : {target_week:,.0f} h  →  Gap = {gap_week:,.0f} h")
print(f"Predicted total next month  : {forecast_month_hours:,.0f} h")
print(f"94 % capacity target month  : {target_month:,.0f} h  →  Gap = {gap_month:,.0f} h")

# ─────────── 7. NEW PROJECTS NEEDED (rough) ────────────────────────────
if "ActualEffort" in df_core.columns:
    avg_proj_week_hours = df_core["ActualEffort"].mean() / 12
else:
    avg_proj_week_hours = 160.0                    # sensible default

new_projects_week  = int(np.ceil(gap_week  / avg_proj_week_hours)) if gap_week  else 0
new_projects_month = int(np.ceil(gap_month / (avg_proj_week_hours*N_WEEKS_IN_MONTH))) if gap_month else 0

print(f"\nNew projects needed ➜  week: {new_projects_week},  month: {new_projects_month}")

# ─────────── 8. EMPLOYEE LOAD TABLE ────────────────────────────────────
emp_load = (
    future_rows[["ResourceEmail", "PredHoursNextMonth"]]
    .rename(columns={"PredHoursNextMonth": "ForecastMonthHrs"})
)
emp_load["Capacity"] = HOURS_PER_WEEK * N_WEEKS_IN_MONTH
emp_load["SpareHrs"] = emp_load["Capacity"] - emp_load["ForecastMonthHrs"]
emp_load = emp_load.sort_values("SpareHrs", ascending=False).reset_index(drop=True)

print("\nTop-10 employee spare capacity:")
print(emp_load.head(10))
